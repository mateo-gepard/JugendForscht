// ==================== CONNECTION ====================
function connect() {
    if (window.location.protocol === 'file:' || !window.location.host) { enableMockMode(); return; }
    var wsUrl = 'ws://' + window.location.host + '/ws';
    try { ws = new WebSocket(wsUrl); } catch(e) { enableMockMode(); return; }
    ws.onopen = function() { setConnected(true); };
    ws.onclose = function() { setConnected(false); setTimeout(connect, 2000); };
    ws.onmessage = function(e) { try { handleMessage(JSON.parse(e.data)); } catch(err) { console.error('Parse error:', err); } };
    ws.onerror = function() {};
}
function enableMockMode() {
    mockMode = true; setConnected(true);
    var bar = document.getElementById('connBar');
    bar.style.background = 'rgba(236,186,78,.1)'; bar.style.color = '#9E7A2A';
    bar.style.borderColor = 'rgba(236,186,78,.18)';
    document.getElementById('connText').textContent = 'Demo';
    showToast('Demo-Modus: Keine VR-Verbindung');
}

// ==================== MESSAGE HANDLING ====================
function handleMessage(data) {
    if (data.type === 'loaded') showToast(data.molecule + ' geladen \u00b7 ' + data.atoms + ' Atome');
    else if (data.type === 'error') showToast('Fehler: ' + data.message);
    else if (data.type === 'status') showToast(data.message);
    else if (data.type === 'tutorial') handleTutorialUpdate(data);
    else if (data.type === 'chirality_result') handleChiralityResult(data);
    else if (data.type === 'quiz_question') handleQuizQuestion(data);
    else if (data.type === 'quiz_answer_result') handleQuizAnswerResult(data);
    else if (data.type === 'quiz_finished') handleQuizFinished(data);
    else if (data.type === 'lorentz_state') handleLorentzState(data);
    else if (data.type === 'swing_state') handleSwingState(data);
    else if (data.type === 'induction_state') handleInductionState(data);
}

    if (ws && ws.readyState === WebSocket.OPEN) { ws.send(JSON.stringify(obj)); return true; }
    showToast('Keine Verbindung'); return false;
}

function handleMockResponse(obj) {
    if (obj.type === 'load' || obj.type === 'iso_load') {

    } else if (obj.type === 'chirality' && obj.action === 'detect') {
        setTimeout(function() {

        }, 800);
    } else if (obj.type === 'isomer' && obj.action === 'mirror') {
        setTimeout(function() { showToast('Enantiomer erzeugt'); }, 600);
    } else if (obj.type === 'isomer' && obj.action === 'overlay') {
        setTimeout(function() { showToast('\u00dcberlagerungstest gestartet'); }, 400);
    } else if (obj.type === 'tutorial') {
        setTimeout(function() {
            if (obj.action==='start') handleMessage({type:'tutorial',status:'started'});
            else if (obj.action==='close') handleMessage({type:'tutorial',status:'closed'});
            else if (obj.action==='continue') handleMessage({type:'tutorial',status:'continued'});
            else if (obj.action==='previous') handleMessage({type:'tutorial',status:'previous'});
        }, 300);
    } else if (obj.type === 'quiz') {
        setTimeout(function() {
            var cat = obj.mode || 'Keilstrich';
            if (obj.action === 'start') {
                mockQuizIndex = 0; mockQuizScore = 0;
                if (cat === 'Keilstrich') handleMessage({type:'quiz_question',questionIndex:0,totalQuestions:2,questionText:'Was zeigt ein ausgef\u00fcllter Keil an?',moleculeName:'methane',category:'Keilstrich',answers:['Bindung nach hinten','Bindung nach vorne','Doppelbindung'],score:0});
                else handleMessage({type:'quiz_question',questionIndex:0,totalQuestions:2,questionText:'Ist Bromchlorfluormethan chiral?',moleculeName:'',category:'Chirality',answers:['Ja, chiral','Nein, achiral'],score:0});
            } else if (obj.action === 'answer') {
                var correct = (obj.answer === 0); if (correct) mockQuizScore++;
                handleMessage({type:'quiz_answer_result',correct:correct,correctIndex:0,selectedIndex:obj.answer,explanation:'Erkl\u00e4rung zur Antwort.',score:mockQuizScore,questionIndex:mockQuizIndex});
            } else if (obj.action === 'next') {
                mockQuizIndex++;
                if (mockQuizIndex >= 2) handleMessage({type:'quiz_finished',score:mockQuizScore,total:2});
                else handleMessage({type:'quiz_question',questionIndex:mockQuizIndex,totalQuestions:2,questionText:'Demo-Frage '+(mockQuizIndex+1),moleculeName:'ethanol',category:quizActiveCategory||'Keilstrich',answers:['Antwort A','Antwort B','Antwort C'],score:mockQuizScore});
            } else if (obj.action === 'end') handleMessage({type:'quiz_finished',score:mockQuizScore,total:2});
        }, 400);
    } else if (obj.type === 'lorentz') {
        setTimeout(function() {
            var mc = mockLorentzCharge || 1;
            if (obj.action==='toggle_charge') mc = -mc;
            else if (obj.action==='set_charge_positive') mc = 1;
            else if (obj.action==='set_charge_negative') mc = -1;
            mockLorentzCharge = mc;
            var st = 'idle';
            if (obj.action==='play') st='running'; else if (obj.action==='pause') st='paused'; else if (obj.action==='reset') st='idle';

// ==================== TUTORIAL ====================
function handleTutorialUpdate(data) {
    if (data.status==='started') { tutorialActive=true;currentUnit=1;waitingForContinue=false;currentUnitName=''; }
    else if (data.status==='closed') { tutorialActive=false;currentUnit=0;waitingForContinue=false;currentUnitName=''; }
    else if (data.status==='continued') { currentUnit++;waitingForContinue=false;if(currentUnit>totalUnits){tutorialActive=false;currentUnit=0;} }
    else if (data.status==='previous') { currentUnit=Math.max(1,currentUnit-1);waitingForContinue=false; }
    else if (data.status==='waitingContinue') { waitingForContinue=true;if(data.unit!==undefined)currentUnit=data.unit+1;if(data.unitName)currentUnitName=data.unitName; }
    updateTutorialUI();
}
function updateTutorialUI() {
    var ring = document.getElementById('progressRing');
    var bar = document.getElementById('progressBar');
    var label = document.getElementById('tutorialLabel');
    var value = document.getElementById('tutorialValue');
    var unitEl = document.getElementById('unitName');
    var idle = document.getElementById('tutorialIdle');
    var active = document.getElementById('tutorialActive');
    var prev = document.getElementById('btnPrev');
    var next = document.getElementById('btnNext');
    var hero = document.querySelector('.tut-hero');
    var nameDisp = document.getElementById('tutNameDisplay');
    var detailDisp = document.getElementById('tutDetail');

    if (tutorialActive) {
        idle.style.display = 'none'; active.style.display = 'block';
        var pct = Math.round((currentUnit/totalUnits)*100);
        bar.style.width = pct + '%';
        ring.style.strokeDashoffset = CIRC * (1 - pct/100);
        value.textContent = currentUnit;
        label.textContent = '/' + totalUnits;
        nameDisp.textContent = currentUnitName || ('Einheit ' + currentUnit);
        if (waitingForContinue) {
            hero.className = 'tut-hero tut-waiting';
            unitEl.textContent = 'BEREIT';
            detailDisp.textContent = 'N\u00e4chste Einheit verf\u00fcgbar';
            next.disabled = false;
        } else {
            hero.className = 'tut-hero tut-active';
            unitEl.textContent = 'L\u00c4UFT';
            detailDisp.textContent = 'Video l\u00e4uft\u2026';
            next.disabled = true;
        }
        prev.disabled = (currentUnit <= 1);
        next.innerHTML = (currentUnit >= totalUnits) ? '\u2713 Fertig' : 'Weiter \u203a';
    } else {
        idle.style.display = 'flex'; active.style.display = 'none';
        hero.className = 'tut-hero';
        value.textContent = '\u2014'; label.textContent = 'Start';
        unitEl.textContent = ''; nameDisp.textContent = 'Keilstrichformel-Tutorial';
        detailDisp.textContent = totalUnits + ' Einheiten';
        ring.style.strokeDashoffset = CIRC; bar.style.width = '0%';
    }
}

function quizNext() { send({type:'quiz',action:'next'}); }
function quizEnd()  { send({type:'quiz',action:'end'}); }
function quizReset() {
    ['quizPanelKeilstrich','quizPanelChirality'].forEach(function(id){var e=document.getElementById(id);e.style.display='none';e.innerHTML='';});
    document.getElementById('btnQuizStartKeilstrich').style.display='block';document.getElementById('btnQuizEndKeilstrich').style.display='none';

function quizAnswer(i) {
    if(quizAnswered) return; quizAnswered=true;
    send({type:'quiz',action:'answer',answer:i});
    document.querySelectorAll('#quizAnswers .nb').forEach(function(b){b.style.pointerEvents='none';b.style.opacity='0.7';});
}
function handleQuizQuestion(data) {
    quizAnswered = false;
    quizActiveCategory = data.category === 'Keilstrich' ? 'Keilstrich' : 'Chirality';
    var c = getQuizContainer(); if(!c) return;
    c.style.display = 'block'; c.innerHTML = buildQuizHTML('active');
    document.getElementById('quizCategory').textContent = data.category==='Keilstrich'?'\u25b2 Keilstrich':'\u262b Chiralit\u00e4t';
    document.getElementById('quizScore').textContent = data.score + ' Punkte';
    document.getElementById('quizProgressLabel').textContent = 'Frage '+(data.questionIndex+1)+' von '+data.totalQuestions;
    document.getElementById('quizProgressBar').style.width = Math.round((data.questionIndex/data.totalQuestions)*100)+'%';
    document.getElementById('quizQuestionText').textContent = data.questionText;
    document.getElementById('quizMolecule').innerHTML = data.moleculeName ? '<svg class="ic"><use href="#i-flask"/></svg> '+data.moleculeName : '';
    var ac = document.getElementById('quizAnswers'); ac.innerHTML='';
    data.answers.forEach(function(a,i){
        var b=document.createElement('button');b.className='nb nb-ghost';b.style.fontSize='14px';
        b.textContent=a;b.onclick=function(){quizAnswer(i);};ac.appendChild(b);
    });
}
function handleQuizAnswerResult(data) {
    document.querySelectorAll('#quizAnswers .nb').forEach(function(b,i){
        b.style.pointerEvents='none';
        if(i===data.correctIndex){b.style.background='var(--ok)';b.style.color='#1a5c30';b.style.opacity='1';b.style.boxShadow='0 4px 14px rgba(92,189,122,.25)';}
        else if(i===data.selectedIndex&&!data.correct){b.style.background='rgba(232,107,95,.12)';b.style.color='var(--err)';b.style.opacity='1';b.style.border='1px solid var(--err)';}
        else{b.style.opacity='0.3';}
    });
    var s=document.getElementById('quizScore');if(s)s.textContent=data.score+' Punkte';
    if(data.explanation){document.getElementById('quizExplanation').style.display='block';document.getElementById('quizExplanationText').textContent=data.explanation;}
    document.getElementById('quizNextRow').style.display='block';
    showToast(data.correct?'Richtig!':'Falsch!');
}
function handleQuizFinished(data) {
    var c=getQuizContainer();if(!c)return;c.innerHTML=buildQuizHTML('finished');
    document.getElementById('btnQuizStartKeilstrich').style.display='block';document.getElementById('btnQuizEndKeilstrich').style.display='none';
    document.getElementById('btnQuizStartChirality').style.display='block';document.getElementById('btnQuizEndChirality').style.display='none';
    var pct=Math.round((data.score/data.total)*100);
    document.getElementById('quizResultScore').textContent=data.score+' von '+data.total+' ('+pct+'%)';
    var ra;
    var icons={80:'i-trophy',60:'i-check',40:'i-search',0:'i-zap'};
    var labels={80:'Ausgezeichnet!',60:'Gut gemacht!',40:'Nicht schlecht.',0:'Weiter \u00fcben!'};
    var tier=pct>=80?80:pct>=60?60:pct>=40?40:0;ra=labels[tier];
    document.getElementById('quizResultEmoji').innerHTML='<svg class="ic" style="width:56px;height:56px;color:var(--warn)"><use href="#'+icons[tier]+'"/></svg>';
    document.getElementById('quizResultRating').textContent=ra;
}

function rBackspace(){
    var inp=document.getElementById('riemannInput'),v=inp.value;
    var funcs=['sqrt(','cbrt(','log(','exp(','sin(','cos(','abs('];
    for(var f=0;f<funcs.length;f++){if(v.endsWith(funcs[f])){inp.value=v.slice(0,-funcs[f].length);return;}}
    if(v.endsWith('pi')){inp.value=v.slice(0,-2);return;}
    inp.value=v.slice(0,-1);
}
