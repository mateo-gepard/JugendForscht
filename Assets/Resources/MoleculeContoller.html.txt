    <script>
        let ws;
        let tutorialActive = false;
        let currentUnit = 0;
        let totalUnits = 11;
        let waitingForContinue = false;
        let currentUnitName = '';

        const statusBadge = document.getElementById('statusBadge');
        const statusText = document.getElementById('statusText');
        const tutorialProgress = document.getElementById('tutorialProgress');
        const tutorialLabel = document.getElementById('tutorialLabel');
        const tutorialValue = document.getElementById('tutorialValue');
        const unitNameEl = document.getElementById('unitName');
        const progressBar = document.getElementById('progressBar');
        const tutorialIdle = document.getElementById('tutorialIdle');
        const tutorialActiveEl = document.getElementById('tutorialActive');
        const btnPrev = document.getElementById('btnPrev');
        const btnNext = document.getElementById('btnNext');

        function connect() {
            const wsUrl = 'ws://' + window.location.host + '/ws';
            ws = new WebSocket(wsUrl);

            ws.onopen = function() {
                statusBadge.className = 'status-badge connected';
                statusText.textContent = 'Verbunden';
            };

            ws.onclose = function() {
                statusBadge.className = 'status-badge';
                statusText.textContent = 'Getrennt';
                setTimeout(connect, 2000);
            };

            ws.onmessage = function(e) {
                try {
                    var data = JSON.parse(e.data);
                    if (data.type === 'loaded') {
                        showToast(data.molecule + ' geladen \u00B7 ' + data.atoms + ' Atome');
                    } else if (data.type === 'error') {
                        showToast('Fehler: ' + data.message);
                    } else if (data.type === 'status') {
                        showToast(data.message);
                    } else if (data.type === 'tutorial') {
                        handleTutorialUpdate(data);
                    }
                } catch (err) {
                    console.error('Parse error:', err);
                }
            };

            ws.onerror = function() {};
        }

        function handleTutorialUpdate(data) {
            if (data.status === 'started') {
                tutorialActive = true;
                currentUnit = 1;
                waitingForContinue = false;
                currentUnitName = '';
                updateTutorialUI();
            } else if (data.status === 'closed') {
                tutorialActive = false;
                currentUnit = 0;
                waitingForContinue = false;
                currentUnitName = '';
                updateTutorialUI();
            } else if (data.status === 'continued') {
                currentUnit++;
                waitingForContinue = false;
                if (currentUnit > totalUnits) {
                    tutorialActive = false;
                    currentUnit = 0;
                }
                updateTutorialUI();
            } else if (data.status === 'previous') {
                currentUnit = Math.max(1, currentUnit - 1);
                waitingForContinue = false;
                updateTutorialUI();
            } else if (data.status === 'waitingContinue') {
                waitingForContinue = true;
                if (data.unit !== undefined) currentUnit = data.unit + 1;
                if (data.unitName) currentUnitName = data.unitName;
                updateTutorialUI();
            } else if (data.step !== undefined) {
                currentUnit = data.step;
                updateTutorialUI();
            }
        }

        function updateTutorialUI() {
            if (tutorialActive) {
                tutorialIdle.style.display = 'none';
                tutorialActiveEl.style.display = 'block';

                var pct = Math.round((currentUnit / totalUnits) * 100);
                progressBar.style.width = pct + '%';
                tutorialLabel.textContent = 'Einheit ' + currentUnit + ' von ' + totalUnits;
                tutorialValue.textContent = currentUnitName || ('Einheit ' + currentUnit);

                if (waitingForContinue) {
                    tutorialProgress.className = 'tutorial-progress waiting';
                    unitNameEl.textContent = 'Bereit f\u00FCr n\u00E4chste Einheit';
                    btnNext.disabled = false;
                } else {
                    tutorialProgress.className = 'tutorial-progress active';
                    unitNameEl.textContent = 'Video l\u00E4uft\u2026';
                    btnNext.disabled = true;
                }

                btnPrev.disabled = (currentUnit <= 1);
                btnNext.innerHTML = (currentUnit >= totalUnits) ? '\u2713 Fertig' : 'Weiter \u203A';
            } else {
                tutorialIdle.style.display = 'flex';
                tutorialActiveEl.style.display = 'none';
                tutorialProgress.className = 'tutorial-progress';
                tutorialLabel.textContent = 'Nicht gestartet';
                tutorialValue.textContent = '\u2014';
                unitNameEl.textContent = '';
                progressBar.style.width = '0%';
            }
        }

        function send(obj) {
            if (ws && ws.readyState === WebSocket.OPEN) {
                ws.send(JSON.stringify(obj));
                return true;
            }
            showToast('Keine Verbindung');
            return false;
        }

        function tutorialStart() { send({ type: 'tutorial', action: 'start' }); }
        function tutorialClose() { send({ type: 'tutorial', action: 'close' }); }
        function tutorialNext() {
            if (currentUnit >= totalUnits) {
                send({ type: 'tutorial', action: 'close' });
            } else {
                send({ type: 'tutorial', action: 'continue' });
            }
        }
        function tutorialPrev() { send({ type: 'tutorial', action: 'previous' }); }

        function loadMolecule(name) {
            if (send({ type: 'load', molecule: name })) {
                showToast('Lade ' + name + '\u2026');
            }
        }

        function searchMolecule() {
            var name = document.getElementById('searchInput').value.trim();
            if (name) {
                loadMolecule(name);
                document.getElementById('searchInput').value = '';
            }
        }

        document.getElementById('searchInput').addEventListener('keypress', function(e) {
            if (e.key === 'Enter') searchMolecule();
        });

        var toastTimer;
        function showToast(msg) {
            var t = document.getElementById('toast');
            t.textContent = msg;
            t.classList.add('visible');
            clearTimeout(toastTimer);
            toastTimer = setTimeout(function() { t.classList.remove('visible'); }, 2500);
        }

        updateTutorialUI();
        connect();
    </script>
