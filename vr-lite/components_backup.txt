function parseSDF(sdfText) {
    const lines = sdfText.split('\n');
    if (lines.length < 5) return null;

    const countsLine = lines[3].trim();
    // V2000: first 3 chars = atom count, next 3 = bond count
    let atomCount, bondCount;
    if (countsLine.includes('V2000') || countsLine.includes('V3000')) {
        const parts = countsLine.trim().split(/\s+/);
        atomCount = parseInt(parts[0]);
        bondCount = parseInt(parts[1]);
    } else {
        const parts = countsLine.trim().split(/\s+/);
        atomCount = parseInt(parts[0]);
        bondCount = parseInt(parts[1]);
    }

    if (isNaN(atomCount) || isNaN(bondCount)) return null;

    const atoms = [];
    const bonds = [];

    for (let i = 0; i < atomCount; i++) {
        const line = lines[4 + i];
        if (!line) continue;
        const p = line.trim().split(/\s+/);
        if (p.length < 4) continue;
        atoms.push({
            id: i,
            x: parseFloat(p[0]),
            y: parseFloat(p[1]),
            z: parseFloat(p[2]),
            element: p[3].replace(/[^A-Za-z]/g, '')
        });
    }

    for (let i = 0; i < bondCount; i++) {
        const line = lines[4 + atomCount + i];
        if (!line) continue;
        const p = line.trim().split(/\s+/);
        if (p.length < 3) continue;
        bonds.push({
            a1: parseInt(p[0]) - 1,
            a2: parseInt(p[1]) - 1,
            type: parseInt(p[2]),
            stereo: p.length >= 4 ? parseInt(p[3]) : 0
        });
    }

    return { atoms, bonds };
}

function centerAndScale(mol, targetSize) {
    targetSize = targetSize || 0.7;
    if (!mol || !mol.atoms.length) return 1;
    let minX=Infinity,maxX=-Infinity,minY=Infinity,maxY=-Infinity,minZ=Infinity,maxZ=-Infinity;
    mol.atoms.forEach(a => {
        if(a.x<minX)minX=a.x;if(a.x>maxX)maxX=a.x;
        if(a.y<minY)minY=a.y;if(a.y>maxY)maxY=a.y;
        if(a.z<minZ)minZ=a.z;if(a.z>maxZ)maxZ=a.z;
    });
    const cx=(minX+maxX)/2, cy=(minY+maxY)/2, cz=(minZ+maxZ)/2;
    const maxDim = Math.max(maxX-minX, maxY-minY, maxZ-minZ, 0.1);
    const scale = targetSize / maxDim;
    mol.atoms.forEach(a => { a.x=(a.x-cx)*scale; a.y=(a.y-cy)*scale; a.z=(a.z-cz)*scale; });
    return scale;
}

const PUBCHEM_BASE = 'https://pubchem.ncbi.nlm.nih.gov/rest/pug';

async function fetchMolecule(name) {
    // Search CID
    const searchUrl = `${PUBCHEM_BASE}/compound/name/${encodeURIComponent(name)}/cids/JSON`;
    const searchResp = await fetch(searchUrl);
    if (!searchResp.ok) throw new Error(`Molek├╝l "${name}" nicht gefunden`);
    const searchData = await searchResp.json();
    const cid = searchData.IdentifierList?.CID?.[0];
    if (!cid) throw new Error(`Keine CID f├╝r "${name}"`);

    // Try 3D first
    let sdfUrl = `${PUBCHEM_BASE}/compound/cid/${cid}/SDF?record_type=3d`;
    let sdfResp = await fetch(sdfUrl);
    if (!sdfResp.ok) {
        // Fallback to 2D
        sdfUrl = `${PUBCHEM_BASE}/compound/cid/${cid}/SDF`;
        sdfResp = await fetch(sdfUrl);
        if (!sdfResp.ok) throw new Error('SDF nicht verf├╝gbar');
    }
    return await sdfResp.text();
}

AFRAME.registerComponent('stereo-renderer', {
    init() {
        this.stereoCamera = new THREE.StereoCamera();
        this.stereoCamera.eyeSep = 0.064; // 64mm interpupillary distance
        this.active = false;
        this._origRender = null;
        this._size = new THREE.Vector2();
        this._doubleTapTimer = 0;
        this._boundDoubleTap = this._onDoubleTap.bind(this);
    },

    activate() {
        if (this.active) return;
        this.active = true;

        const sceneEl = this.el;
        const renderer = sceneEl.renderer;
        const stereoCamera = this.stereoCamera;
        const sceneObj = sceneEl.object3D;
        const size = this._size;

        // Store original render
        this._origRender = renderer.render.bind(renderer);
        const origRender = this._origRender;

        // Override renderer.render for stereo
        renderer.render = function(scene, camera) {
            // Only apply stereo to the main scene render
            if (scene !== sceneObj) {
                return origRender(scene, camera);
            }

            renderer.getSize(size);
            const w = size.x, h = size.y;

            stereoCamera.update(camera);

            // Save state
            const autoClear = renderer.autoClear;
            renderer.autoClear = false;
            renderer.clear();
            renderer.setScissorTest(true);

            // Left eye
            renderer.setViewport(0, 0, w / 2, h);
            renderer.setScissor(0, 0, w / 2, h);
            origRender(scene, stereoCamera.cameraL);

            // Right eye
            renderer.setViewport(w / 2, 0, w / 2, h);
            renderer.setScissor(w / 2, 0, w / 2, h);
            origRender(scene, stereoCamera.cameraR);

            // Restore
            renderer.setScissorTest(false);
            renderer.setViewport(0, 0, w, h);
            renderer.autoClear = autoClear;
        };

        // UI: show divider, exit button
        document.body.classList.add('stereo-active');
        document.getElementById('stereo-divider').style.display = 'block';
        document.getElementById('stereo-exit').style.display = 'block';
        document.getElementById('stereo-label').style.display = 'block';

        // Fullscreen
        const el = document.documentElement;
        if (el.requestFullscreen) el.requestFullscreen().catch(() => {});
        else if (el.webkitRequestFullscreen) el.webkitRequestFullscreen();

        // Lock landscape
        try { screen.orientation.lock('landscape-primary').catch(() => {}); } catch(e) {}

        // Double-tap to exit
        document.addEventListener('touchend', this._boundDoubleTap);

        console.log('[StereoVR] Activated ÔÇö split-screen rendering');
    },

    deactivate() {
        if (!this.active) return;
        this.active = false;

        const renderer = this.el.renderer;
        if (this._origRender) {
            renderer.render = this._origRender;
            this._origRender = null;
        }
        renderer.setScissorTest(false);
        renderer.getSize(this._size);
        renderer.setViewport(0, 0, this._size.x, this._size.y);

        // UI: hide divider, exit button
        document.body.classList.remove('stereo-active');
        document.getElementById('stereo-divider').style.display = 'none';
        document.getElementById('stereo-exit').style.display = 'none';
        document.getElementById('stereo-label').style.display = 'none';

        // Exit fullscreen
        if (document.fullscreenElement) document.exitFullscreen().catch(() => {});
        else if (document.webkitFullscreenElement) document.webkitExitFullscreen();

        // Unlock orientation
        try { screen.orientation.unlock(); } catch(e) {}

        document.removeEventListener('touchend', this._boundDoubleTap);

        console.log('[StereoVR] Deactivated ÔÇö normal rendering');
    },

    _onDoubleTap(e) {
        const now = Date.now();
        if (now - this._doubleTapTimer < 400) {
            this.deactivate();
            document.getElementById('info-bar').style.display = 'flex';
        }
        this._doubleTapTimer = now;
    }
});

function getColor(el) { return ELEMENT_COLORS[el] || '#CCCCCC'; }
function getRadius(el) { return ELEMENT_RADII[el] || 0.08; }

function renderMolecule(mol, containerEl) {
    // Clear existing
    while (containerEl.firstChild) containerEl.removeChild(containerEl.firstChild);

    if (!mol) return;
    centerAndScale(mol, 1.2);

    // Wrapper for rotation
    const wrapper = document.createElement('a-entity');
    wrapper.setAttribute('auto-rotate', 'speed: 0.15');
    wrapper.id = 'mol-wrapper';

    // Render atoms
    mol.atoms.forEach((atom, i) => {
        const sphere = document.createElement('a-sphere');
        sphere.setAttribute('position', `${atom.x.toFixed(4)} ${atom.y.toFixed(4)} ${atom.z.toFixed(4)}`);
        sphere.setAttribute('radius', getRadius(atom.element));
        const color = getColor(atom.element);
        sphere.setAttribute('material', `color:${color}; metalness:0.2; roughness:0.4; emissive:${color}; emissiveIntensity:0.15`);
        sphere.setAttribute('class', 'interactive');
        sphere.setAttribute('atom-interact', `element:${atom.element}; idx:${i}`);
        sphere.setAttribute('segments-width', 24);
        sphere.setAttribute('segments-height', 16);
        wrapper.appendChild(sphere);
    });

    // Render bonds
    mol.bonds.forEach(bond => {
        const a1 = mol.atoms[bond.a1];
        const a2 = mol.atoms[bond.a2];
        if (!a1 || !a2) return;

        if (bond.type === 1 && bond.stereo === 0) {
            createBondCylinder(a1, a2, 0.025, '#888888', wrapper);
        } else if (bond.type === 2) {
            createDoubleBond(a1, a2, wrapper);
        } else if (bond.type === 3) {
            createTripleBond(a1, a2, wrapper);
        } else if (bond.stereo === 1) {
            // Wedge bond (solid)
            createWedgeBond(a1, a2, wrapper);
        } else if (bond.stereo === 6) {
            // Dash bond
            createDashBond(a1, a2, wrapper);
        } else {
            createBondCylinder(a1, a2, 0.025, '#888888', wrapper);
        }
    });

    containerEl.appendChild(wrapper);
}

function createBondCylinder(a1, a2, radius, color, parent, offset) {
    offset = offset || {x:0, y:0, z:0};
    const p1 = new THREE.Vector3(a1.x + offset.x, a1.y + offset.y, a1.z + offset.z);
    const p2 = new THREE.Vector3(a2.x + offset.x, a2.y + offset.y, a2.z + offset.z);
    const mid = p1.clone().add(p2).multiplyScalar(0.5);
    const dir = p2.clone().sub(p1);
    const len = dir.length();
    dir.normalize();

    const Y = new THREE.Vector3(0, 1, 0);
    const quat = new THREE.Quaternion().setFromUnitVectors(Y, dir);
    const euler = new THREE.Euler().setFromQuaternion(quat, 'YXZ');

    const cyl = document.createElement('a-cylinder');
    cyl.setAttribute('position', `${mid.x.toFixed(4)} ${mid.y.toFixed(4)} ${mid.z.toFixed(4)}`);
    cyl.setAttribute('rotation', `${r2d(euler.x)} ${r2d(euler.y)} ${r2d(euler.z)}`);
    cyl.setAttribute('height', len.toFixed(4));
    cyl.setAttribute('radius', radius);
    cyl.setAttribute('material', `color:${color}; metalness:0.3; roughness:0.5; emissive:${color}; emissiveIntensity:0.05`);
    cyl.setAttribute('segments-radial', 8);
    parent.appendChild(cyl);
    return cyl;
}

function createDoubleBond(a1, a2, parent) {
    const dir = new THREE.Vector3(a2.x-a1.x, a2.y-a1.y, a2.z-a1.z).normalize();
    let up = new THREE.Vector3(0, 1, 0);
    let perp = new THREE.Vector3().crossVectors(dir, up).normalize();
    if (perp.lengthSq() < 0.01) {
        up.set(1, 0, 0);
        perp.crossVectors(dir, up).normalize();
    }
    const off = 0.03;
    createBondCylinder(a1, a2, 0.018, '#888888', parent, {x:perp.x*off, y:perp.y*off, z:perp.z*off});
    createBondCylinder(a1, a2, 0.018, '#888888', parent, {x:-perp.x*off, y:-perp.y*off, z:-perp.z*off});
}

function createTripleBond(a1, a2, parent) {
